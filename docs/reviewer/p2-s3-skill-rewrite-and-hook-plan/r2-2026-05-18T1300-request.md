import re
from tasktool.allocate import _id_from_filename  # if missing, replicate locally

_FILENAME_ID_RE = re.compile(r"^\d{4}-\d{2}-\d{2}-(?P<id>[Pp]\d+(?:\.[SsXx]\d+[a-z]?)?(?:\.[Tt]\d+)?)-")

def collect_known_ids(p) -> set[str]:
    ids: set[str] = set()
    for ph in p.phases:
        ids.add(ph.id)
        for sl in ph.slices:
            ids.add(f"{ph.id}.{sl.id}")
            ids.add(sl.id)
            for t in sl.tasks:
                ids.add(f"{ph.id}.{sl.id}.{t.id}")
        # also archived phases survive
    for ph in getattr(p, "archived_phases", []) or []:
        ids.add(ph.id if hasattr(ph, "id") else ph["id"])
    for x in p.cross_cutting:
        ids.add(x.id)
    return ids

def validate_orphan_filenames(p, paths) -> list[str]:
    known = collect_known_ids(p)
    findings: list[str] = []
    for path in paths:
        name = Path(path).name
        m = _FILENAME_ID_RE.match(name)
        if not m:
            continue
        raw = m.group("id").upper()
        # Accept fully-qualified or short forms. Both `P2.S3` and `S3` (under context) should be looked up.
        if raw in known or raw.split(".", 1)[-1] in known:
            continue
        findings.append(f"{path}: filename references ID {raw} but no matching row in tasklist.json")
    return findings

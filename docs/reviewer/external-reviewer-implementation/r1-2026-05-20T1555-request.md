  131	def strip_prompt_echo(text: str) -> str:
  132	    """Remove any superstar-prompt-sentinel-delimited region from `text`.
  133	
  134	    Handles three cases beyond the simple full-block case:
  135	    - End marker present but no start marker → delete from start of stream
  136	      through (and including) the end marker. Models a tail-truncated echo
  137	      where the beginning was capped off but the end marker survived.
  138	    - Start marker present but no end marker → delete from the start marker
  139	      to end of stream. Models a head-truncated echo.
  140	    - Multiple full blocks → all removed.
  141	    """
  142	    if not text:
  143	        return text
  144	    out = text
  145	    # Repeatedly strip full blocks first (greedy non-overlapping).
  146	    while True:
  147	        s = out.find(PROMPT_SENTINEL_START)
  148	        e = out.find(PROMPT_SENTINEL_END)
  149	        if s != -1 and e != -1 and e > s:
  150	            out = out[:s] + out[e + len(PROMPT_SENTINEL_END):]
  151	            continue
  152	        break
  153	    # Truncated-end case: end marker without a preceding start marker.
  154	    e = out.find(PROMPT_SENTINEL_END)
  155	    if e != -1 and out.find(PROMPT_SENTINEL_START) == -1:
  156	        out = out[e + len(PROMPT_SENTINEL_END):]
  157	    # Truncated-start case: start marker without a following end marker.
  158	    s = out.find(PROMPT_SENTINEL_START)
  159	    if s != -1 and out.find(PROMPT_SENTINEL_END) == -1:
  160	        out = out[:s]
  161	    return out

  164	def cap_with_elision(text: str, max_bytes: int = 80 * 1024) -> str:
  165	    """Cap `text` to ~max_bytes, keeping head + tail with an elision marker.
  166	
  167	    Returns the original text unchanged if under the cap. Otherwise returns
  168	    the first 60% + a marker + the last 40% of `max_bytes`. Bytes count is
  169	    on the encoded UTF-8 length; for purely ASCII content this equals
  170	    character count.
  171	    """
  172	    if not text:
  173	        return text
  174	    raw = text.encode("utf-8")
  175	    if len(raw) <= max_bytes:
  176	        return text
  177	    head_bytes = int(max_bytes * 0.6)
  178	    tail_bytes = max_bytes - head_bytes
  179	    head = raw[:head_bytes].decode("utf-8", errors="ignore")
  180	    tail = raw[-tail_bytes:].decode("utf-8", errors="ignore")
  181	    elided = len(raw) - head_bytes - tail_bytes
  182	    marker = f"\n\n[… {elided} bytes elided to fit cap of {max_bytes} bytes …]\n\n"
  183	    return head + marker + tail

  212	class _StateLock:
  213	    """Context manager acquiring fcntl.LOCK_EX on the state-file lock companion.
  214	    Both readers and writers acquire this lock so writes serialize across
  215	    processes (spec line 79)."""
  216	
  217	    def __init__(self) -> None:
  218	        self._fd: int | None = None
  219	
  220	    def __enter__(self) -> "_StateLock":
  221	        _ensure_state_parent_dir()
  222	        lock_path = _state_lock_path()
  223	        # O_RDWR | O_CREAT, mode 0o600 — owner-only lock companion.
  224	        self._fd = os.open(str(lock_path), os.O_RDWR | os.O_CREAT, 0o600)
  225	        fcntl.flock(self._fd, fcntl.LOCK_EX)
  226	        return self
  227	
  228	    def __exit__(self, exc_type, exc, tb) -> None:
  229	        if self._fd is not None:
  230	            try:
  231	                fcntl.flock(self._fd, fcntl.LOCK_UN)
  232	            finally:
  233	                os.close(self._fd)
  234	                self._fd = None

  237	def load_state() -> dict:
  238	    """Read the reviewer state file. Fails open: missing/corrupt → empty state.
  239	    Acquires fcntl.flock(LOCK_EX) on the lock companion so reads see a
  240	    consistent snapshot relative to in-flight writers."""
  241	    path = state_file_path()
  242	    with _StateLock():
  243	        if not path.exists():
  244	            return {"schema_version": 1, "limits": {}}
  245	        try:
  246	            data = json.loads(path.read_text(encoding="utf-8"))
  247	            if not isinstance(data, dict) or data.get("schema_version") != 1 or not isinstance(data.get("limits"), dict):
  248	                raise ValueError("schema mismatch")
  249	            return data
  250	        except (json.JSONDecodeError, ValueError, OSError) as e:
  251	            print(f"WARNING: reviewer-state.json at {path} unreadable ({e}); treating as empty", file=sys.stderr)
  252	            return {"schema_version": 1, "limits": {}}

  255	def save_state(state: dict) -> None:
  256	    """Atomically write the reviewer state file. Uses flock on a lock companion
  257	    + tmp-then-rename. Creates parent dir with mode 0o700 on first write."""
  258	    path = state_file_path()
  259	    tmp_path = path.with_suffix(path.suffix + ".tmp")
  260	    payload = json.dumps(state, indent=2, sort_keys=True)
  261	    with _StateLock():
  262	        with open(tmp_path, "w", encoding="utf-8") as f:
  263	            f.write(payload)
  264	            f.flush()
  265	            os.fsync(f.fileno())
  266	        os.replace(tmp_path, path)

  296	def update_state(mutator) -> dict:
  297	    """Atomic read-modify-write under the state lock. `mutator(state)` may
  298	    mutate the dict in place or return a new dict. Returns the final state.
  299	    This is the recommended path for any caller that does a read followed by
  300	    a dependent write, since plain load_state + save_state pairs are not
  301	    atomic across processes."""
  302	    with _StateLock():
  303	        state = _load_state_locked()
  304	        result = mutator(state)
  305	        if result is not None and isinstance(result, dict):
  306	            state = result
  307	        _save_state_locked(state)
  308	        return state

  312	    """Return the limit entry for the given reviewer if it's still active.

  357	def resolve_reviewer_provider(
  358	    *,
  359	    reviewer_provider: str,
  360	    caller_provider: str,
  361	    reviewer_cmd: str | None,
  362	    env: dict | None = None,

  366	    caller = caller_provider or detect_caller_provider(env)

  371	    if explicit_cmd:

  374	    if provider == "auto":
  375	        if caller == "claude":
  376	            provider = "codex"
  377	        elif caller == "codex":

  386	        raise ProviderResolutionError(f"Unknown reviewer provider: {provider}")
  387	    if provider == "custom":
  388	        raise ProviderResolutionError("provider=custom requires AGENT_REVIEWER_CMD or --reviewer-cmd")

  413	def _user_patterns_from_env() -> list[tuple[str, re.Pattern]]:
  414	    raw = os.environ.get("AGENT_REVIEWER_RATE_LIMIT_PATTERNS", "")
  415	    if not raw:
  416	        return []
  417	    pairs = []
  418	    for chunk in raw.split(";"):
  419	        chunk = chunk.strip()
  420	        if "=" not in chunk:
  421	            continue
  422	        name, pattern = chunk.split("=", 1)
  423	        try:
  424	            pairs.append((name.strip(), re.compile(pattern.strip(), re.IGNORECASE | re.DOTALL)))
  425	        except re.error:
  426	            print(f"WARNING: invalid user rate-limit pattern '{name}': skipping", file=sys.stderr)
  427	    return pairs

  430	def detect_rate_limit(stderr_text: str) -> tuple[bool, "dt.datetime | None", "str | None"]:
  431	    """Inspect reviewer stderr for a rate-limit signature.
  432	    Returns (matched, reset_at_local, pattern_name)."""
  433	    patterns = RATE_LIMIT_BUILTIN_PATTERNS + _user_patterns_from_env()
  434	    for name, pat in patterns:
  435	        m = pat.search(stderr_text)
  436	        if m:
  437	            time_group = m.group(1) if m.groups() else None
  438	            reset_at = _parse_reset_time(time_group) if time_group else _fallback_reset_time()
  439	            return True, reset_at, name
  440	    return False, None, None

  453	    hours = int(os.environ.get("AGENT_REVIEWER_LIMIT_FALLBACK_HOURS", "4"))

  457	def _parse_reset_time(s: str) -> "dt.datetime":
  458	    """Parse a clock time (HH:MM with optional AM/PM, or 24-hour) as local time.
  459	    If the parsed time is in the past relative to now, add one day."""
  460	    s = (s or "").strip()
  461	    hour, minute = None, None
  462	    m = _TIME_RE_AMPM.match(s)
  463	    if m:
  464	        hour, minute = int(m.group(1)), int(m.group(2))
  465	        suffix = m.group(3).upper()
  466	        if suffix == "PM" and hour < 12:
  467	            hour += 12
  468	        elif suffix == "AM" and hour == 12:
  469	            hour = 0
  470	    else:
  471	        m = _TIME_RE_24H.match(s)
  472	        if m:
  473	            hour, minute = int(m.group(1)), int(m.group(2))
  474	    if hour is None or not (0 <= hour <= 23) or not (0 <= minute <= 59):
  475	        return _fallback_reset_time()
  476	    now = _now_local()
  477	    candidate = now.replace(hour=hour, minute=minute, second=0, microsecond=0)
  478	    if candidate <= now:
  479	        candidate += dt.timedelta(days=1)
  480	    return candidate

  494	    whichever comes first.
  495	
  496	    Uses only the four known ``_BUDGET_SECTIONS`` patterns as boundaries
  497	    (not any ``\\n## `` heading) so that nested sub-headings inside a
  498	    section (e.g. ``### git diff base..HEAD`` inside "Changes since prior
  499	    round") do not prematurely terminate the section.
  500	    """
  501	    import re
  502	    candidates = []
  503	    for _, pattern, _ in _BUDGET_SECTIONS:
  504	        m = re.search(pattern, text[section_start:])
  505	        if m:
  506	            candidates.append(section_start + m.start())
  507	    e = text.find(PROMPT_SENTINEL_END, section_start)
  508	    if e != -1:
  509	        candidates.append(e)
  510	    if not candidates:
  511	        return len(text)
  512	    return min(candidates)
  513	
  514	
  515	def apply_budget(text: str, budget_chars: int) -> str:
  516	    """Trim prunable sections in priority order until `text` fits the budget.
  517	
  518	    Preserved (never trimmed):
  519	      - Sentinel markers
  520	      - Chain summary table (`## Review chain summary`)
  521	      - Review-mode preamble + REVIEW_PROMPT contract
  522	
  523	    Pruning order (lowest priority dropped first):
  524	      1. Target Preview        → 80 → 40 → 0 lines
  525	      2. Diff body             → 50 KB → 12 KB → 0
  526	      3. Resolution body       → 20 KB → 8 KB → 2 KB
  527	      4. Prior findings body   → 40 KB → 16 KB → 8 KB
  528	
  529	    Appends a `<!-- budget-applied: ... -->` HTML comment immediately before
  530	    the end sentinel summarising trims.
  531	
  532	    Note: the final string may exceed `budget_chars` by up to ~200 bytes — the
  533	    trim loop fits content to the budget, then appends a diagnostic note.
  534	    """
  535	    import re
  536	    if len(text) <= budget_chars:
  537	        return text
  538	
  539	    out = text
  540	    trim_log: list[str] = []
  541	    for name, pattern, levels in _BUDGET_SECTIONS:
  542	        if len(out) <= budget_chars:
  543	            break
  544	        m = re.search(pattern, out)
  545	        if not m:
  546	            continue
  547	        section_start = m.end()
  548	        for level_bytes in levels:
  549	            if len(out) <= budget_chars:
  550	                break
  551	            section_end = _find_section_end(out, section_start)
  552	            section_body = out[section_start:section_end]
  553	            if level_bytes == 0:
  554	                replacement = f"\n[{name} dropped to fit budget]\n"
  555	            else:
  556	                replacement = "\n" + cap_with_elision(section_body, max_bytes=level_bytes) + "\n"
  557	            if len(replacement) >= len(section_body):
  558	                continue
  559	            out = out[:section_start] + replacement + out[section_end:]
  560	            trim_log.append(f"{name}:{level_bytes}")
  561	
  562	    note = (
  563	        f"\n<!-- budget-applied: budget={budget_chars} "
  564	        f"trims=[{','.join(trim_log)}] final_size={len(out)} -->\n"
  565	    )
  566	    end_idx = out.rfind(PROMPT_SENTINEL_END)
  567	    if end_idx != -1:
  568	        out = out[:end_idx] + note + out[end_idx:]
  569	    else:
  570	        out = out + note
  571	    return out

Each round writes a `request.md` and `response.md` pair under a per-document chain folder so the iteration history is durable and committable.
    9	
   10	**Bridge command.** `external-reviewer` is the global canonical review-chain bridge command. It is installed by `skills/external-review/install.sh` and delegates to `skills/external-review/scripts/external-reviewer.py` in the active Superstar checkout. When the source script is under `$HOME`, the generated shim stores a literal `$HOME/...` path so it does not pin a machine-specific `/home/simon/...` string while still expanding to the checkout at runtime. Do not run or copy a full repo-local `scripts/external-reviewer.py` bridge. Existing repos may keep a tiny compatibility shim at that path only so old handoffs continue to delegate to the global command.
   11	
   12	**Announce at start:** "I'm using the external-review skill to run a `<kind>` review on `<target>`."
   13	
   14	## When to use
   15	
   16	Four checkpoints, mapped to `--kind`:
   17	
   18	| Stage                          | `--kind`      | Triggered by                                                              |
   19	|--------------------------------|---------------|---------------------------------------------------------------------------|
   20	| Spec written, before plan      | `spec`        | `[[writing-plans]]` after the spec is saved, before drafting the plan     |
   21	| Plan written, before execution | `plan`        | `[[writing-plans]]` after the plan is saved, before handing off to execute|
   22	| Slice complete                 | `post-slice`  | `[[subagent-driven-development]]` after a slice's tasks all close         |
   23	| Phase complete                 | `post-phase`  | `[[subagent-driven-development]]` after the last slice of a phase closes  |
   24	
   25	`design`, `implementation`, and `other` are valid for ad-hoc reviews and do not gate the main workflow.
   26	
   27	## When NOT to use
   28	
   29	- Mid-implementation single-commit asks — use `[[requesting-internal-review]]` instead.
   30	- WIP / unstable targets — the reviewer needs a stable file on disk.
   31	- The user wants the coordinating agent itself to review — that is a different skill.

   33	## Configuration
   34	
   35	By default the bridge chooses the opposite reviewer provider from the caller:
   36	
   37	| Caller | Default reviewer |
   38	|---|---|
   39	| Claude | Codex |
   40	| Codex | Claude |
   41	
   42	Provider selection is controlled by `--reviewer-provider auto|codex|claude|custom` or `AGENT_REVIEWER_PROVIDER`. Caller detection is controlled by `--caller-provider auto|claude|codex|unknown` or `AGENT_REVIEWER_CALLER`. If both are `auto` and the caller cannot be detected, the bridge fails closed and asks for an explicit provider or command.
   43	
   44	The reviewer command is still overrideable via `AGENT_REVIEWER_CMD` or `--reviewer-cmd`. Any explicit reviewer command is treated as `custom` and bypasses provider auto-selection. Custom wrappers are responsible for their own sandboxing.
   45	
   46	The default command remains `reviewer-agent`. The safe wrapper contract is:
   47	
   48	- reviewed repo is readable but not writable;
   49	- `AGENT_REVIEWER_SCRATCH_DIR` is writable and short-lived;
   50	- `AGENT_REVIEWER_RESPONSE_DIR` is writable for final-message handoff;
   51	- wrappers may write machine-readable sidecars such as `reviewer-metadata.json`,
   52	  `codex-events.jsonl`, or `claude-output.json` under
   53	  `AGENT_REVIEWER_RESPONSE_DIR`; the review text contract remains stdout plus
   54	  the persisted response artifact;
   55	- wrappers must not use Codex `--dangerously-bypass-approvals-and-sandbox` or Claude `--dangerously-skip-permissions` unless the operator has supplied an external OS sandbox and chosen a custom command.
   56	- Codex currently uses `disk-full-read-access`, which may expose files outside the repo for reading. This fork accepts that read-side risk to keep the write-side mitigation simple.
   57	
   58	The command may be:
   59	
   60	- A bare executable (`reviewer-agent`) — the prompt is supplied per `--prompt-transport` (`stdin` | `arg` | `file`, default `stdin`).
   61	- A template with placeholders (`{prompt_file}`, `{prompt_text}`, `{target_file}`, `{kind}`, `{chain_dir}`, `{round}`, `{previous_response}`, `{resolution_file}`, `{session_file}`, `{repo_root}`, `{response_dir}`, `{scratch_dir}`, `{request_file}`) — substituted and run through the shell. Env vars are authoritative; placeholders are derived convenience values.
   62	
   63	The bridge exports `AGENT_REVIEWER_REPO_ROOT`, `AGENT_REVIEWER_CHAIN_DIR`, `AGENT_REVIEWER_REQUEST_FILE`, `AGENT_REVIEWER_RESPONSE_DIR`, `AGENT_REVIEWER_SCRATCH_DIR`, `AGENT_REVIEWER_TARGET_FILE`, `AGENT_REVIEWER_KIND`, `AGENT_REVIEWER_ROLE`, and `AGENT_REVIEWER_SWEEP_INDEX` for every reviewer process. `AGENT_REVIEWER_SWEEP_INDEX` is always set: empty for primary, numeric for sweeps. These env vars are authoritative; command placeholders are convenience sugar derived from the same values.
   64	
   65	Scratch directories are owner-only and normally removed by the bridge. If a process is killed before cleanup, remove stale dirs with:
   66	
   67	```bash
   68	find "${TMPDIR:-/tmp}" -maxdepth 1 -type d -name 'superstar-reviewer-*' -mtime +1 -prune -exec rm -rf -- {} +
   69	```
   70	
   71	If `reviewer-agent` is missing, `[[project-setup]]` will offer to install/configure it. If the command emits no `Overall verdict`, treat the round as `revise` and ask the reviewer to honour the response contract on the next round.
   72	
   73	- `--incremental-budget-chars` (default `400000`) sets a target cap on assembled prompt size for incremental rounds. The prompt is pruned in priority order — target preview, diff body, resolution body, prior findings body — toward the target; sentinel markers, chain summary, and finding-ID lists are never trimmed. The final size may exceed the target by ~150 bytes due to the appended `<!-- budget-applied: ... -->` diagnostic note.
   74	
   75	## How a round runs
   76	
   77	For `post-slice` and `post-phase`, run a scope preflight before invoking the reviewer:
   78	
   79	```bash
   80	git status --short
   81	```
   82	
   83	If the status includes unrelated dirty files, setup/migration artifacts, legacy path moves, untracked vendored reviewer scripts, copied chain output from unrelated work, unrelated reviewer chains, unrelated tasklist mutations, files from another slice, or any dirty files outside the slice/phase scope, stop and resolve the boundary first. This is a hard blocker, not reviewer discretion: do not invoke `post-slice` or `post-phase` review against an ambiguous artifact set.
   84	
   85	Use the requested `--work-id` to judge scope before review:
   86	
   87	- `docs/reviewer/<this-chain>/...` for the current work is allowed; reviewer chain folders for another slice/phase are blockers.
   88	- `docs/tasklist.json` changes are allowed only when they correspond to the current slice/phase closeout; allocation/status changes for other work are blockers.
   89	- Dirty implementation files must belong to the current slice/phase evidence set; files from another slice must move to that slice's worktree or be merged separately before review.
   90	- A normal `main`/`master` checkout is planning/review-only by default; post-slice/post-phase review should run from the slice/phase implementation worktree unless the human partner explicitly opted out of isolation in the current turn.
   91	
   92	A reviewer is expected to fail a boundary review when the artifact set is ambiguous, but the coordinator should block before spending reviewer time.
   93	
   94	```bash
   95	external-reviewer review \
   96	    --kind <spec|plan|post-slice|post-phase> \
   97	    --file <path/to/target.md> \
   98	    --work-id <P2.S3 | P2>   # required for post-slice / post-phase
   99	    [--context <path>]... \
  100	    [--review-depth thorough] \
  101	    [--reviewer-provider auto|codex|claude|custom] \
  102	    [--caller-provider auto|claude|codex|unknown] \
  103	    [--incremental-budget-chars 400000] \
  104	    --emit json
  105	```
  106	
  107	- Output folder: `docs/reviewer/<target-stem-no-date>[-<work-id-dotless>]-<kind>/`
  108	- Round number, base ref, and prior verdict are read from `chain.json` in the chain folder.
  109	- Each round emits `r{N}-{timestamp}-request.md` and `r{N}-{timestamp}-response.md`. When `--review-depth thorough` or `exhaustive` runs sweep reviewers, filenames become `r{N}-{ts}-primary-*.md` and `r{N}-{ts}-sweep{K}-*.md`, plus a `r{N}-merged-findings.md`.
  110	- `--emit json` returns the structured payload described in "Reading the response". Always use `--emit json` from this skill — agents consume the JSON, not paths or human prose.
  111	
  112	The command **blocks** until the reviewer exits (default `--timeout 900`). Run it in the **foreground**. Do not background it, do not poll the chain folder, do not retry in a loop.
  113	
  114	**Prompt transport.** The script defaults to `stdin` for every bare reviewer command. This matches the bundled `reviewer-agent` wrapper and avoids `ARG_MAX` failures when prompts grow large. Override with `--prompt-transport {stdin|arg|file}` or `AGENT_REVIEWER_TRANSPORT` only when a custom reviewer backend cannot accept stdin.
  115	
  116	## Failure handling
  117	
  118	When the configured reviewer command exits non-zero, the round is recorded as a **process failure**, not as a verdict:
  119	
  120	- The persisted response file is a short stub (≤ 8 KB total): header, status, and the sentinel-stripped tail of the reviewer's stderr capped at 4 KB. No stdout is written.
  121	- `chain.json` records `status: "failed"`, `returncode: <rc>`, `verdict: null`, `verdict_valid: false` on both the round entry and the per-reviewer entry.
  122	- For `post-slice` / `post-phase`, the next round's resolution-required gate is **bypassed** with a stderr notice. A process failure has no findings to resolve; the next round is a re-attempt of the same review, not a fix-and-re-review.
  123	- The next round's preamble walks backward past `status: "failed"` (and legacy `status: "unknown"`) rounds and embeds the merged-findings from the most recent `status: "ok"` round, prefixed with a `Note: rounds N..K were process failures...; skipped.` line. If no successful prior round exists, only the chain summary table is embedded.
  124	
  125	**Sentinel-wrapped prompts.** Every prompt is wrapped in `<!-- superstar-prompt:start -->` / `<!-- superstar-prompt:end -->` markers. If a reviewer echoes the prompt on stdout or stderr, the markers let the script strip the echo before persisting to disk, eliminating the recursive prompt-bloat class.
  126	
  127	### Multi-reviewer truth (sweeps)
  128	
  129	When `--review-depth thorough` or `exhaustive` runs sweeps alongside the primary:
  130	
  131	| Primary | Sweeps | Top-level `status` | `verdict_valid` | `merged_verdict` | Process exit |
  132	|---|---|---|---|---|---|
  133	| ok | all ok | `ok` | per merged | computed | `0` |
  134	| ok | some failed | `ok` | per merged (ok reviewers only) | computed from ok | `0` |
  135	| ok | all failed | `ok` | per primary | primary's verdict | `0` |
  136	| failed | any/all | `failed` | `false` | `null` | primary's returncode |
  137	
  138	Failed sweeps are excluded from merged-findings and do not flip the top-level status.
  139	
  140	## Rate-limit handling
  141	
  142	When the reviewer's provider rate-limits the configured command (e.g. codex usage cap, Claude API quota), the script detects the failure mode distinctly from a generic crash and stops to ask the operator.
  143	
  144	**Exit code 8** signals "reviewer rate-limited; pick a recovery path." On exit 8 the script emits this JSON on stdout:
  145	
  146	```json
  147	{
  148	  "rate_limited": true,
  149	  "reviewer_cmd": "<basename>",
  150	  "reset_at":    "<ISO local time>",
  151	  "reset_source": "regex:<pattern-name>",
  152	  "chain":  "<chain folder name>",
  153	  "round":  <int>,
  154	  "request_path": "<absolute path>",
  155	  "raw_stderr_tail": "<last 2 KB of reviewer stderr>"
  156	}
  157	```
  158	
  159	Persistent state lives at `~/.config/superstar/reviewer-state.json` (override via `AGENT_REVIEWER_STATE_FILE` or `--state-file`). Subsequent invocations against any chain refuse to spawn until `reset_at` passes.
  160	
  161	### The recovery menu
  162	
  163	On exit 8 the coordinator MUST present this menu via `AskUserQuestion` (no auto-pick):
  164	
  165	| Option | Mechanism |
  166	|---|---|
  167	| **Manual approve** | Coordinator collects a one-line note, then runs `external-reviewer manual-approve --kind X --file Y --work-id Z --note "..."`. Writes a synthetic round with `status: "manual-approved"`, `verdict: "ready"`. Chain advances. |
  168	| **Schedule retry** | Coordinator invokes the **harness-level `schedule` skill** to register a one-shot routine at `reset_at + 5 min` re-invoking the same `review` command. If the harness lacks `schedule`, falls back to printing an `at`/`cron`-suitable command for the operator. Current chain gate pauses. |
  169	| **Human bridge** | Coordinator prints `r{N}-request.md` path. Operator obtains a response from an external reviewer (web UI, manual reading, etc.) and either pastes the text in chat or provides a local file path. Coordinator runs `external-reviewer ingest-response --kind X --file Y --work-id Z (--from-paste FILE \| --from-link PATH)`. Writes the response with status `human-bridged`. |
  170	| **Hold** | Do nothing. Exit the current gate. State persists; next session sees the same limit. |
  171	
  172	Repeated refusals against the **same chain** while the limit is open do NOT append new rounds — they coalesce onto the head rate-limited round via `last_refused_at` / `refused_at[]` (capped at 20).
  173	
  174	### Status semantics
  175	
  176	A `status: "rate-limited"` round is treated symmetrically with `status: "failed"`:
  177	- The resolution-required gate is bypassed for the next round.
  178	- `build_incremental_preamble` walks back past it to find the last `ok` round.
  179	- It is excluded from `merged_verdict` and `write_merged_findings` aggregation.
  180	
  181	Manual-approved (`status: "manual-approved"`) and human-bridged (`status: "human-bridged"`) rounds carry real verdicts and pass through the existing gating machinery unchanged.

  195	The JSON output (always use `--emit json`) is the source of truth. Agents MUST consult:

  208	"""Normalized tracker model for the timeline generator.
  209	
  210	This is the ONLY module that knows the docs/tasklist.json schema. If a shared
  211	tracker-model module is later extracted from tasktool, replace this module's
  212	internals; TimelineItem is the seam consumed by extract/render/timeline.
  213	"""
  214	
  215	from __future__ import annotations
  216	
  217	import datetime as dt
  218	from dataclasses import dataclass, field
  219	
  220	EPOCH_PLACEHOLDER = "1970-01-01"
  221	TERMINAL_STATUSES = {"done", "cancelled"}
  222	START_STATUSES = {"in_progress", "started"}
  223	
  224	
  225	def parse_tracker_date(raw):
  226	    """ISO 'YYYY-MM-DD' or 'YYYY-MM-DDTHH:MM[:SS]' -> (datetime|None, precision)."""
  227	    if not raw or raw == EPOCH_PLACEHOLDER:
  228	        return None, "day"
  229	    if "T" in raw:
  230	        return dt.datetime.fromisoformat(raw), "minute"
  231	    d = dt.date.fromisoformat(raw)
  232	    return dt.datetime(d.year, d.month, d.day), "day"
  233	
  234	
  235	@dataclass
  236	class DateValue:
  237	    when: dt.datetime | None = None
  238	    precision: str = "day"      # "day" | "minute"
  239	    source: str = "field"       # "field" | "replay" | "override"
  240	
  241	
  242	@dataclass
  243	class TimelineItem:
  244	    key: str                    # "P21", "P21.S4", "X13"
  245	    kind: str                   # "phase" | "slice" | "x"
  246	    parent: str | None
  247	    title: str
  248	    status: str
  249	    display_title: str | None = None
  250	    created: DateValue = field(default_factory=DateValue)
  251	    started: DateValue = field(default_factory=DateValue)
  252	    closed: DateValue = field(default_factory=DateValue)
  253	    excluded: bool = False
  254	
  255	    def label(self):
  256	        return self.display_title or self.title
  257	
  258	
  259	def _date_value(obj, name):
  260	    when, precision = parse_tracker_date(obj.get(name))
  261	    return DateValue(when, precision, "field")
  262	
  263	
  264	def _item(key, kind, parent, obj):
  265	    return TimelineItem(
  266	        key=key, kind=kind, parent=parent,
  267	        title=obj.get("title") or key,
  268	        status=obj.get("status", "unknown"),
  269	        created=_date_value(obj, "created"),
  270	        started=_date_value(obj, "started"),
  271	        closed=_date_value(obj, "closed"),
  272	    )
  273	
  274	
  275	def items_from_project(doc):
  276	    """Walk a project-shaped dict (live tasklist.json or an archived
  277	    '## Full phase JSON' block) into TimelineItem records."""
  278	    items = []
  279	    for p in doc.get("phases", []):
  280	        pk = p["id"]
  281	        items.append(_item(pk, "phase", None, p))
  282	        for s in p.get("slices", []):
  283	            items.append(_item(f"{pk}.{s['id']}", "slice", pk, s))
  284	    for c in doc.get("cross_cutting", []):
  285	        items.append(_item(c["id"], "x", None, c))
  286	    return items
  287	
  288	
  289	def item_from_cross(obj):
  290	    """A single archived cross-cutting item object
  291	    (from a '## Full cross-cutting JSON' block)."""
  292	    return _item(obj["id"], "x", None, obj)
  293	
  294	
  295	def collect(live_doc, archive_project_docs, archive_x_objects):
  296	    """Merge all sources into one item list. First occurrence of a key wins,
  297	    and live is read first, so live data shadows any stale archive copy."""
  298	    seen, items = set(), []
  299	    sources = [items_from_project(live_doc)]
  300	    sources += [items_from_project(d) for d in archive_project_docs]
  301	    sources += [[item_from_cross(o)] for o in archive_x_objects]
  302	    for source in sources:
  303	        for it in source:
  304	            if it.key not in seen:
  305	                seen.add(it.key)
  306	                items.append(it)
  307	    return items
  308	```

  425	````python
  426	"""Read tracker data out of a repo: live file, archive JSON blocks, git replay."""
  427	
  428	from __future__ import annotations
  429	
  430	import json
  431	import re
  432	import subprocess
  433	from dataclasses import dataclass, field
  434	from pathlib import Path
  435	
  436	TRACKER = "docs/tasklist.json"
  437	
  438	_PHASE_BLOCK_RE = re.compile(
  439	    r"^## Full phase JSON.*?^```json\n(.*?)^```", re.S | re.M)
  440	_CROSS_BLOCK_RE = re.compile(
  441	    r"^## Full cross-cutting JSON.*?^```json\n(.*?)^```", re.S | re.M)
  442	
  443	
  444	def git(repo, *args, check=True):
  445	    proc = subprocess.run(["git", "-C", str(repo), *args],
  446	                          capture_output=True, text=True)
  447	    if check and proc.returncode != 0:
  448	        raise SystemExit(f"timeline: git {' '.join(args)} failed: "
  449	                         f"{proc.stderr.strip()}")
  450	    return proc.stdout
  451	
  452	
  453	def repo_root(path):
  454	    return Path(git(path, "rev-parse", "--show-toplevel").strip())
  455	
  456	
  457	def read_live(repo):
  458	    p = Path(repo) / TRACKER
  459	    if not p.exists():
  460	        raise SystemExit(f"timeline: {p} not found — not a tasktool project")
  461	    return json.loads(p.read_text())
  462	
  463	
  464	def read_archives(repo):
  465	    """-> (project_docs, x_objects, warnings).
  466	
  467	    Reads both '## Full phase JSON' blocks (a project-shaped object whose
  468	    `phases` array holds the archived phase) and '## Full cross-cutting JSON'
  469	    blocks (a single item object). Files with neither block (pure-legacy
  470	    markdown) are ignored — they are backfill.py's input, not ours.
  471	    """
  472	    project_docs, x_objects, warnings = [], [], []
  473	    arch = Path(repo) / "docs" / "archived-tasks"
  474	    files = sorted(arch.glob("*.md")) if arch.is_dir() else []
  475	    for f in files:
  476	        text = f.read_text()
  477	        pm = _PHASE_BLOCK_RE.search(text)
  478	        cm = _CROSS_BLOCK_RE.search(text)
  479	        try:
  480	            if pm:
  481	                project_docs.append(json.loads(pm.group(1)))
  482	            elif cm:
  483	                x_objects.append(json.loads(cm.group(1)))
  484	        except json.JSONDecodeError as e:
  485	            warnings.append(f"{f.name}: unparseable JSON block: {e}")
  486	    return project_docs, x_objects, warnings
  487	````

### docs/specs/2026-06-06-X29-timeline-design.md

    1	# X29 — Visual work-history timeline generator (`tools/timeline`)
    2	
    3	**Date:** 2026-06-06
    4	**Status:** draft
    5	**Kind:** cross-cutting tool (human-facing; zero agent-context footprint)
    6	
    7	## Purpose
    8	
    9	A script that generates a browser-viewable visual timeline of completed work for any
   10	tasktool-managed project, intended as a visual aid when discussing progress with
   11	non-technical people. A vertical spine down the centre of a scrollable page represents
   12	project time from start to latest activity; phases and slices are the major and minor
   13	nodes on that spine, placed at their completion dates/times.
   14	
   15	## Constraints (binding)
   16	
   17	- **Zero agent-context overhead.** No skill, hook, CLAUDE.md entry, or tasktool help
   18	  line references this tool. It is invoked only by humans or on explicit request in
   19	  interactive sessions.
   20	- **Zero dependencies.** Python 3 stdlib only; git accessed via `subprocess`. No
   21	  third-party packages, matching the repo's zero-dependency philosophy.
   22	- **Generic.** Works on any repo with `docs/tasklist.json` (current tasktool schema),
   23	  not just this one. Primary target project today: `multistore`.
   24	- **Static output.** A single self-contained HTML file (inline CSS/JS/data). Opens via
   25	  `file://`, shareable as an attachment, no server.
   26	
   27	## Placement & architecture
   28	
   29	```
   30	tools/timeline/
   31	  timeline.py    # CLI entry
   32	  extract.py     # git replay + live tasklist.json + archive JSON blocks → raw events
   33	  model.py       # the ONE schema-aware module → normalized TimelineItem records
   34	  render.py      # TimelineItem records → self-contained HTML
   35	  backfill.py    # run-once legacy migrator (separate command; never invoked by timeline.py)
   36	```
   37	
   38	Sibling to `tools/tasktool`, **not** part of it (tasktool's CLI surface is unchanged).
   39	`model.py` is the deliberate seam: it is the only module that knows the tracker schema.
   40	If a shared tracker-model module is later extracted from tasktool, only `model.py`'s
   41	internals are replaced; `extract.py` and `render.py` consume `TimelineItem` records and
   42	do not move.
   43	
   44	### CLI
   45	
   46	```
   47	python3 tools/timeline/timeline.py [--repo PATH] [-o timeline.html] [--show-x] [--overrides PATH]
   48	python3 tools/timeline/backfill.py [--repo PATH] [--write]
   49	```
   50	
   51	- `--repo` defaults to the current working directory's repo root (`git rev-parse --show-toplevel`).
   52	- `-o` defaults to `timeline.html` in the current directory.
   53	- `--show-x` sets the X-item toggle's initial state to on (data is embedded either way).
   54	- `--overrides` defaults to `docs/timeline-overrides.json` in the target repo if present.
   55	- `backfill.py` is dry-run by default, printing a unified diff; `--write` applies it.
   56	
   57	## Data model
   58	
   59	```python
   60	TimelineItem:
   61	  key            # "P21", "P21.S4", "X13"
   62	  kind           # phase | slice | x
   63	  parent         # phase key for slices, None otherwise
   64	  title          # verbatim tracker title
   65	  display_title  # optional override relabel; renderer prefers it when present
   66	  status         # done | cancelled | ready | in_progress | blocked ...
   67	  created, started, closed   # each: (datetime|None, precision, source)
   68	                             # precision: day | minute
   69	                             # source:    field | replay | override
   70	```
   71	
   72	## Date resolution (precedence)
   73	
   74	For each of `created`/`started`/`closed` on each item:
   75	
   76	1. **Overrides file** (`docs/timeline-overrides.json`, optional) — always wins. Schema:
   77	
   78	   ```json
   79	   {
   80	     "items": {
   81	       "P14":    { "started": "2026-05-20" },
   82	       "P21.S2": { "display_title": "Quiet-launch controls" },
   83	       "X12":    { "exclude": true }
   84	     }
   85	   }
   86	   ```
   87	
   88	   Values: ISO date or datetime for the three date fields; `display_title` string;
   89	   `exclude` boolean. Unknown keys in an item entry are an error (fail loud, not silent).
   90	
   91	2. **Tracker JSON fields** — the authoritative *date* (day precision). Sources: the live
   92	   `docs/tasklist.json` plus every archive file's fenced JSON block under
   93	   `docs/archived-tasks/` — `## Full phase JSON` blocks for archived phases (a full
   94	   project-shaped object whose `phases` array holds the phase) **and** `## Full
   95	   cross-cutting JSON` blocks for archived X-items (a single item object). Both shapes
   96	   are required reading; dropping the cross-cutting blocks would leave the X toggle
   97	   mostly empty on mature projects. The placeholder `1970-01-01` is treated as absent.
   98	   Where two archive files exist for the same phase (legacy + tasktool re-archive), the
   99	   one with a parseable JSON block wins; pure-legacy markdown files are ignored by
  100	   `timeline.py` (they are `backfill.py`'s input, not the renderer's).
  101	
  102	3. **Git replay** — walk every commit touching `docs/tasklist.json` oldest→newest
  103	   (`git log --reverse --format=%H %ct -- docs/tasklist.json`), parse the file at each
  104	   revision (`git show SHA:docs/tasklist.json`), and record status transitions and date
  105	   field changes per item with the commit timestamp. Replay:
  106	   - upgrades a field date to **minute precision** when the replay-observed transition
  107	     falls on the same calendar day;
  108	   - fills a field that is null in the tracker (e.g. missing `started`: first transition
  109	     to `in_progress`/`started`; missing phase `closed` is *not* invented — an open phase
  110	     renders as open);
  111	   - **ignores transitions observed at an item's first-appearance commit when the item
  112	     appears already terminal** (import artifacts — e.g. the 2026-05-18 multistore
  113	     migration commit where P1–P12 arrived `done`).
  114	
  115	Validated against multistore: 226 commits replay with zero parse failures, yielding
  116	minute-precision lifecycles for 108 items.
  117	
  118	### Derived phase span
  119	
  120	- Phase start for rendering = `started` if present, else earliest slice `started`, else
  121	  phase `created`.
  122	- Phase end = `closed` if present, else open (strand runs to the bottom edge labelled
  123	  with the generation date).
  124	- **Close-only items** (no resolvable start or create — e.g. legacy P1 with
  125	  `created: 1970-01-01`, `started: null`, no slices): render as a close node only — a
  126	  zero-length span with the hollow close ring and label, no strand segment. This is a
  127	  supported display mode, not an error; backfill/overrides can later supply a start.
  128	
  129	## Visual specification
  130	
  131	Validated interactively against real multistore data (P20–P23).
  132	
  133	- **Spine braid.** Vertical centre spine; one coloured strand per open phase. When N
  134	  phases are open concurrently, N strands sit side-by-side. When no phase is open, a
  135	  dotted grey strand bridges the gap with an "N quiet days" label. Phase colours come
  136	  from a fixed palette cycled by phase number (stable across regenerations).
  137	- **Nodes.** Phase start = filled disc in phase colour + bold title and start date.
  138	  Phase close = hollow ring in phase colour (grey ring for cancelled phases) + "PNN
  139	  complete · date" label. Slice completion = small filled disc in phase colour.
  140	- **Cards.** Each slice gets an info card tinted with the parent phase colour
  141	  (background tint + border), showing title (or `display_title`) and completion
  142	  date/time. Day-precision dates show no time-of-day (never a fake "00:00").
  143	  During solo stretches cards alternate left/right; during overlaps each open phase
  144	  owns one side so a track reads as a column. With 3+ concurrent phases the braid
  145	  gains strands; cards keep phase colours for attribution.
  146	- **Click-to-expand.** Clicking a card expands it in place: full verbatim title, item
  147	  ID, started/closed datetimes with precision markers, computed duration.
  148	- **Time scale: proportional with guard rails.** Vertical distance is proportional to
  149	  elapsed time, subject to (a) a minimum spacing between adjacent nodes — burst days
  150	  expand locally to stay readable — and (b) a maximum rendered height for empty
  151	  stretches — long gaps compress to a capped dotted segment with the quiet-days label.
  152	- **Cancelled items.** A cancelled phase is omitted unless it has ≥1 `done` slice; when
  153	  shown it renders normally (grey close ring), with only its completed slices.
  154	  Cancelled slices are always omitted.
  155	- **X-items.** First-class: always extracted and embedded. Rendered as neutral slate
  156	  nodes/cards on the spine at their completion time (they have no parent phase colour).
  157	  An in-page toggle shows/hides them instantly without regeneration; `--show-x` only
  158	  sets the toggle's initial state. Open/never-completed X-items are not rendered.
  159	- **Header.** Project name, overall date span, totals (phases completed, slices
  160	  completed), generation timestamp, and a colour legend of phases.
  161	
  162	## Legacy backfill (`backfill.py`)
  163	
  164	One-time, per-project migration for items that predate the tracker. It exists so that
  165	`timeline.py` never grows legacy parsing paths.
  166	
  167	- **Input:** legacy archive markdown under `docs/archived-tasks/` (headings of the form
  168	  `## S1 — title ✅ \`DONE 2026-04-29\``) and the repo's commit subjects.
  169	- **Recovers:** slice titles and close dates from the markdown; phase/slice `started`
  170	  dates from first-commit-mention mining (`\bP(\d+)(?:[.\-]?S(\d+))?\b` over subjects),
  171	  cross-checked against the previous phase's close for the sequential legacy era.
  172	- **Writes:** recovered slices and dates into the corresponding tasktool archive file's
  173	  `## Full phase JSON` block — the canonical location `tasktool unarchive` understands.
  174	  Phase-level fields already present are never overwritten.
  175	- **Safety:** dry-run by default with a unified diff; `--write` applies; the human
  176	  reviews and commits. Never invoked by `timeline.py`.
  177	
  178	Multistore evidence: legacy P1–P12 have reliable close dates in archives; commit mining
  179	dates P2–P12 starts cleanly; P1 has no commit mentions, so its `started` will need a
  180	manual override entry (or remain start-unknown, rendering from its close date only).
  181	
  182	## Error handling
  183	
  184	- Not a git repo / no `docs/tasklist.json` → clear fatal error.
  185	- A historical revision that fails JSON parsing → skipped with a stderr warning; replay
  186	  continues (validated: 0/226 failures on multistore, but schema drift must not be fatal).
  187	- Shallow clone → replay covers what exists; items degrade to day precision with a
  188	  one-line warning.
  189	- Unknown schema fields → ignored (tolerant reader); unknown override keys → fatal.
  190	- Items with no resolvable dates at all → listed in a stderr summary so gaps are
  191	  visible, omitted from the render. No silent drops.
  192	
  193	## Testing
  194	
  195	`tools/timeline/tests/` (pytest, mirroring tasktool's convention). **`tools/timeline/tests`
  196	must be added to `testpaths` in the repo's `pyproject.toml`** so the default
  197	`python3 -m pytest` gate discovers it — otherwise "all tests pass" is vacuously true.
  198	
  199	- **Replay:** transition detection, minute-precision upgrade, null-fill, import-artifact
  200	  suppression — against a fixture git repo built by the test (tmpdir, scripted commits).
